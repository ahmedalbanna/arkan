## أمثلة على الاختبارات

### اختبار Flow بسيط
```typescript
it('should generate text successfully', async () => {
  // Arrange
  const input = {
    prompt: 'اكتب مقالة عن الذكاء الاصطناعي',
    length: 'medium' as const,
    style: 'professional' as const,
    language: 'ar' as const,
  };

  // Act
  const result = await generateText(input);

  // Assert
  expect(result).toBeDefined();
  expect(result.content).toBeDefined();
  expect(result.wordCount).toBeGreaterThan(0);
  expect(result.readingTime).toBeGreaterThan(0);
});
```

### اختبار معالجة الأخطاء
```typescript
it('should handle AI generation errors', async () => {
  // Arrange
  const input = { prompt: 'test prompt' };
  (ContextAnalyzer.analyzeContext as jest.Mock).mockRejectedValue(
    new Error('AI generation failed')
  );

  // Act & Assert
  await expect(generateText(input)).rejects.toThrow('AI generation failed');
});
```

### اختبار الأداء
```typescript
it('should complete within reasonable time', async () => {
  // Arrange
  const input = { prompt: 'test prompt' };

  // Act
  const startTime = Date.now();
  const result = await generateText(input);
  const endTime = Date.now();

  // Assert
  expect(result).toBeDefined();
  expect(endTime - startTime).toBeLessThan(5000); // أقل من 5 ثوان
});
```
